<?php

class RegistrarVacanteRequest extends FormRequest

    public function authorize(): bool
    {
        return true;

    protected function prepareForValidation(): void
    {
        $datos = $this->all();

        foreach (['titulo', 'descripcion', 'ubicacion'] as $campo) {
            if (
                array_key_exists($campo, $datos)
                && is_string($datos[$campo])
            ) {
                $datos[$campo] = trim($datos[$campo]);
            }
        }

        if (
            array_key_exists('ubicacion', $datos)
            && $datos['ubicacion'] === ''
        ) {
            $datos['ubicacion'] = null;
        }

        // Requisitos y beneficios: se recortan los espacios y se descartan las filas vacías
        foreach (['requisitos', 'beneficios'] as $lista) {
            if (array_key_exists($lista, $datos) && is_array($datos[$lista])) {
                $datos[$lista] = array_values(array_filter(
                    array_map(
                        fn ($item) => is_string($item) ? trim($item) : $item,
                        $datos[$lista]
                    ),
                    fn ($item) => $item !== ''
                ));
            }
        }

        $this->replace($datos);
    }

===== PAGAR VACANTE REQUEST =====
<?php

namespace App\Http\Requests\Pagos;

use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Support\Carbon;
use Illuminate\Validation\Validator;

/**
 * Datos de la tarjeta del pago simulado.
 * Se acepta cualquier tarjeta con número válido (algoritmo de Luhn), vencimiento vigente y CVV: el pago siempre se aprueba.
 * El monto no se recibe: lo calcula el servidor. El número completo y el CVV nunca se guardan.
 */
class PagarVacanteRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    protected function prepareForValidation(): void
    {
        if ($this->has('titular') && is_string($this->titular)) {
            $this->merge([
                'titular' => preg_replace('/\s+/u', ' ', trim($this->titular)),
            ]);
        }
    }

    public function rules(): array
    {
        return [
            'titular' => [
                'required',
                'string',
                'max:100',
            ],

            'numero_tarjeta' => [
                'bail',
                'required',
                'string',
                'digits:16',
            ],

            'vencimiento' => [
                'bail',
                'required',
                'string',
                'regex:/^(0[1-9]|1[0-2])\/\d{2}$/',
            ],

            'cvv' => [
                'required',
                'string',
                'digits:3',
            ],
        ];
    }

    public function after(): array
    {
        return [
            function (Validator $validator): void {
                if ($validator->errors()->has('numero_tarjeta')) {
                    return;
                }

                if (! $this->cumpleLuhn($this->input('numero_tarjeta'))) {
                    $validator->errors()->add('numero_tarjeta', 'El número de tarjeta no es válido.');
                }
            },

            function (Validator $validator): void {
                if ($validator->errors()->has('vencimiento')) {
                    return;
                }

                // La tarjeta sirve hasta el último día del mes indicado
                $vence = Carbon::createFromFormat('!m/y', $this->input('vencimiento'))->endOfMonth();

                if ($vence->isPast()) {
                    $validator->errors()->add('vencimiento', 'La tarjeta está vencida.');
                }
            },
        ];
    }

    public function messages(): array
    {
        return [
            'titular.required' => 'El titular es obligatorio.',
            'titular.string' => 'El titular debe ser una cadena de texto.',
            'titular.max' => 'El titular no debe superar los 100 caracteres.',

            'numero_tarjeta.required' => 'El número de tarjeta es obligatorio.',
            'numero_tarjeta.string' => 'El número de tarjeta debe enviarse como texto.',
            'numero_tarjeta.digits' => 'El número de tarjeta debe tener 16 dígitos.',

            'vencimiento.required' => 'El vencimiento es obligatorio.',
            'vencimiento.string' => 'El vencimiento debe enviarse como texto.',
            'vencimiento.regex' => 'El vencimiento debe tener el formato MM/AA.',

            'cvv.required' => 'El CVV es obligatorio.',
            'cvv.string' => 'El CVV debe enviarse como texto.',
            'cvv.digits' => 'El CVV debe tener 3 dígitos.',
        ];
    }

    // Dígito verificador de las tarjetas: desde la derecha se duplica uno de cada dos dígitos y la suma debe ser múltiplo de 10
    private function cumpleLuhn(string $numero): bool
    {
        if (! preg_match('/^[0-9]{16}$/D', $numero)) {
            return false;
        }

        $suma = 0;
        $duplicar = false;

        for ($i = strlen($numero) - 1; $i >= 0; $i--) {
            $digito = (int) $numero[$i];

            if ($duplicar) {
                $digito *= 2;

                if ($digito > 9) {
                    $digito -= 9;
                }
            }

            $suma += $digito;
            $duplicar = ! $duplicar;
        }

        return $suma % 10 === 0;
    }
}

<?php

namespace App\Http\Resources\Api\V1;

use App\Models\Empleo;
use App\Models\PagoVacante;
use App\Models\RevisionVacante;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

/**
 * Vacante vista por su empresa.
 *
 * Relaciones: nivelPuesto, areaEstudio, habilidades, revision y pagos.
 *
 * @mixin Empleo
 */
class VacanteEmpresaResource extends JsonResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'titulo' => $this->titulo,
            'descripcion' => $this->descripcion,
            'requisitos' => $this->requisitos ?? [],
            'beneficios' => $this->beneficios ?? [],

            'nivel_puesto' => $this->whenLoaded(
                'nivelPuesto',
                fn () => [
                    'id' => $this->nivel_puesto_id,
                    'nombre' => $this->nivelPuesto?->nombre,
                ]
            ),

            'area_profesional' => $this->whenLoaded(
                'areaEstudio',
                fn () => [
                    'id' => $this->area_estudio_id,
                    'nombre' => $this->areaEstudio?->nombre,
                ]
            ),

            'tipo_empleo' => $this->tipo_empleo,
            'ubicacion' => $this->ubicacion,

            // Conservar NULL: ausencia de salario no equivale a cero.
            'salario_min' => $this->salario_min === null
                ? null
                : (float) $this->salario_min,

            'salario_max' => $this->salario_max === null
                ? null
                : (float) $this->salario_max,

            'fecha_inicio_vigencia' => $this->fecha_inicio_vigencia
                ?->toDateString(),

            'fecha_fin_vigencia' => $this->fecha_fin_vigencia
                ?->toDateString(),

            'max_contrataciones' => $this->numero_vacantes,

            'habilidades' => $this->whenLoaded(
                'habilidades',
                fn () => $this->habilidades
                    ->map(fn ($habilidad) => [
                        'id' => $habilidad->id,
                        'nombre' => $habilidad->nombre,
                        'tipo_habilidad' => $habilidad->tipo_habilidad,
                        'requisito' => $habilidad->pivot->tipo_requisito,
                    ])
                    ->values()
            ),

            'estado' => [
                'general' => $this->estadoGeneral(),
                'revision' => RevisionVacante::estadoDe($this->revision),
                'motivo_no_aprobacion' => $this->revision?->decision === RevisionVacante::NO_APROBADA
                    ? $this->revision->motivo
                    : null,
                'pago' => PagoVacante::estadoDe($this->pagos),
                'disponibilidad' => $this->estadoDisponibilidad(),
                'vigente' => $this->estaVigente(),
                'publicada_en' => $this->publicadaEn(),
            ],

            'postulaciones_count' => $this->whenCounted(
                'postulaciones',
                fn ($total) => (int) $total
            ),

            'postulaciones_semana_count' => $this->whenCounted(
                'postulaciones_semana',
                fn ($total) => (int) $total
            ),

            'creado_en' => $this->created_at,
            'eliminado_en' => $this->deleted_at,
        ];
    }
}
